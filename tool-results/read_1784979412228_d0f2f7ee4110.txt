    95→function formatExpiryBadge(expiry: string | null) {
    96→  if (!expiry) return null;
    97→  const d = new Date(expiry);
    98→  if (Number.isNaN(d.getTime())) return null;
    99→  const now = new Date();
   100→  const diffMs = d.getTime() - now.getTime();
   101→  const days = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
   102→  const label = d.toISOString().slice(0, 10);
   103→  if (days < 0) {
   104→    return { label, tone: "red" as const, days };
   105→  }
   106→  if (days <= 30) {
   107→    return { label, tone: "amber" as const, days };
   108→  }
   109→  return { label, tone: "default" as const, days };
   110→}

   265→      loadProducts();

   267→      toast.error("Network error");
   268→    } finally {
   269→      setSaving(false);
   270→    }

  1010→function ProductWizard({ open, onOpenChange, categories, onDone, editProduct }: ProductWizardProps) {
  1011→  const [step, setStep] = React.useState(1);

  1016→  // Shared

  1019→  const [image, setImage] = React.useState<string | null>(null);
  1020→  const [expiryDate, setExpiryDate] = React.useState("");
  1021→  const [manufacturingDate, setManufacturingDate] = React.useState("");
  1022→
  1023→  // Piece info
  1024→  const [pieceBarcode, setPieceBarcode] = React.useState("");

  1026→  const [pieceSalePrice, setPieceSalePrice] = React.useState("");
  1027→  const [pieceWholesalePrice, setPieceWholesalePrice] = React.useState("");
  1028→  const [pieceShopkeeperPrice, setPieceShopkeeperPrice] = React.useState("");

  1037→  const [boxSalePrice, setBoxSalePrice] = React.useState("");

  1102→      setPieceCostPrice(autoPieceCost ? autoPieceCost.toFixed(2) : "");
  1103→    }

  1106→  async function saveProducts() {
  1107→    setSaving(true);
  1108→    try {
  1109→      if (productType === "piece") {
  1110→        const body = {
  1111→          name, barcode: pieceBarcode, categoryId: categoryId || null,
  1112→          costPrice: Number(pieceCostPrice) || 0,
  1113→          salePrice: Number(pieceSalePrice) || 0,
  1114→          wholesalePrice: Number(pieceWholesalePrice) || 0,
  1115→          shopkeeperPrice: Number(pieceShopkeeperPrice) || 0,
  1116→          unit: "piece", stock: Number(pieceStock) || 0,
  1117→          minStock: Number(pieceMinStock) || 0,
  1118→          expiryDate: expiryDate || null,
  1119→          manufacturingDate: manufacturingDate || null,
  1120→          hasBarcode: true, active: true, image,
  1121→        };
  1122→        const url = editId ? `/api/products/${editId}` : "/api/products";
  1123→        const method = editId ? "PUT" : "POST";
  1124→        const res = await fetch(url, { method, headers: { "Content-Type": "application/json" }, body: JSON.stringify(body) });
  1125→        const data = await res.json();
  1126→        if (!res.ok) {
  1127→          const msg = data.error || "Failed";
  1128→          toast.error(msg.includes("barcode") ? "This barcode already exists!" : msg);
  1129→          setSaving(false); return;
  1130→        }
  1131→        toast.success(editId ? "Product updated!" : `${name} added!`);
  1132→      } else {
  1133→        const boxBody = {
  1134→          name: editId ? name : `${name} (Box)`, barcode: boxBarcode, categoryId: categoryId || null,
  1135→          costPrice: Number(boxCostPrice) || 0, salePrice: Number(boxSalePrice) || 0,
  1136→          wholesalePrice: Number(boxWholesalePrice) || 0, shopkeeperPrice: Number(boxShopkeeperPrice) || 0,
  1137→          unit: "piece", stock: totalPieces || Number(pieceStock) || 0,
  1138→          minStock: Number(pieceMinStock) || 0,
  1139→          expiryDate: expiryDate || null, manufacturingDate: manufacturingDate || null,
  1140→          hasBarcode: true, active: true, image,
  1141→          packBarcode: pieceBarcode, packQuantity: Number(piecesPerBox) || 0, packPrice: Number(boxSalePrice) || 0,
  1142→        };
  1143→        const boxUrl = editId ? `/api/products/${editId}` : "/api/products";
  1144→        const boxMethod = editId ? "PUT" : "POST";
  1145→        const boxRes = await fetch(boxUrl, { method: boxMethod, headers: { "Content-Type": "application/json" }, body: JSON.stringify(boxBody) });
